```rust
// Full FederatedServiceRegistry integration
pub async fn register_service(&self, request: ServiceRegistrationRequest) 
    -> Result<ServiceRegistrationResponse, SongbirdError> 
{
    let service_id = Uuid::new_v4();
    let (addr, port) = parse_endpoint(&request.endpoint)?;
    
    let registration = ServiceRegistration {
        service_id,
        capabilities: request.capabilities,
        metadata: request.metadata,
        endpoint: request.endpoint,
        address: addr,
        port,
        last_heartbeat: chrono::Utc::now(),
    };
    
    self.service_registry.register_local(registration).await?;
    Ok(ServiceRegistrationResponse { service_id })
}
```

#### 2. Capability-Based Discovery with AND Logic
```rust
// Multi-capability intersection-based matching
for capability in &query.capabilities {
    let services = self.service_registry
        .find_by_capability(capability).await;
    
    if matching_services.is_empty() {
        matching_services = services;
    } else {
        // Intersection: services must have ALL capabilities
        let service_ids: HashSet<_> = 
            services.iter().map(|s| s.service_id.clone()).collect();
        matching_services.retain(|s| service_ids.contains(&s.service_id));
    }
}

// Apply metadata filters
matching_services.retain(|svc| {
    query.filters.iter().all(|(key, value)| {
        svc.metadata.get(key) == Some(value)
    })
});
```
