**Evolution Strategy**:
```rust
// FROM: Hardcoded
const API_PORT: u16 = 8080;

// TO: Capability-based
pub async fn api_port() -> Result<u16> {
    // 1. Environment variable
    if let Ok(port) = env::var("BEARDOG_API_PORT") {
        return port.parse();
    }
    // 2. Capability discovery
    if let Some(cap) = discover_capability("api").await? {
        return Ok(cap.port);
    }
    // 3. Service mesh query
    if let Some(port) = query_service_mesh("beardog-api").await? {
        return Ok(port);
    }
    // 4. Documented fallback (logged)
    log::info!("Using fallback port 8080");
    Ok(8080)
}
```
